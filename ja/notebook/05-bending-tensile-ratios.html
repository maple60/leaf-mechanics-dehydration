# Load data
df_list <- list()
df_list[["bending"]] <- read.csv("outputs/tables/bending.csv")
df_list[["tensile"]] <- add_tensile_stiffness(read.csv("outputs/tables/tensile.csv"))
df_list[["merged"]] <- merge(
  df_list[["bending"]][, c(
    "spp",
    "leaf_side",
    "ID",
    "EI",
    "E.x",
    "water_content"
  )],
  df_list[["tensile"]][, c("spp", "ID1", "EA", "E.x", "water_content")],
  all = TRUE,
  suffixes = c("_bending", "_tensile"),
  by.x = c("spp", "ID"),
  by.y = c("spp", "ID1")
)

df <- df_list[["merged"]]
df <- df[df$water_content_bending >= 60 & df$water_content_tensile >= 60, ] # filter data for water content >= 60%

df$RWC_mean <- (df$water_content_bending + df$water_content_tensile) / 2 # calculate mean RWC
df$stiffness_ratio <- df$EI / df$EA # calculate ratio of EI
df$E_ratio <- df$E.x_bending / df$E.x_tensile # calculate ratio of E

df$stiffness_ratio <- df$stiffness_ratio * 10^3 # convert unit of EI

spp <- c(
  "ubayuri",
  "youshuyamagobou",
  "yurinoki",
  "niseakashia",
  "aoki",
  "nezumimochi"
) # species names
names(spp) <- c(
  "Cardiocrinum_cordatum",
  "Phytolacca_americana",
  "Liriodendron_tulipifera",
  "Robinia_pseudoacacia",
  "Aucuba_japonica",
  "Ligustrum lucidum"
)
pred_range <- c(60, 100) # Prediction range for RWC
xlim <- c(100, 60) # x-axis limits
mar <- c(5, 5, 0.5, 0.5) # Margins


# Plot GAM with confidence intervals ----
output_dir <- "outputs/figures/figure_03"
if (!dir.exists(output_dir)) {
  dir.create(output_dir, recursive = TRUE)
}

# (A) - (F) RWC vs bending EI / tensile EA
lapply(seq_along(spp), function(i) {
  sp <- spp[i]
  df_sp <- df[df$spp == sp, ]
  df_adaxial <- droplevels(subset(df_sp, leaf_side == "adaxial"))
  df_abaxial <- droplevels(subset(df_sp, leaf_side == "abaxial"))
  x <- "RWC_mean"
  y <- "stiffness_ratio"
  output_path <- file.path(
    output_dir,
    paste0(LETTERS[i], "_RWC_vs_stiffness_ratio_", names(spp)[i], ".svg")
  )
  svg(output_path, bg = "transparent")
  # Plot adaxial side
  plot_gam_ci(
    data = df_adaxial,
    x = x,
    y = y,
    side = "adaxial",
    ylim = gam_ci_ylim(df_sp, x, y, pred_range),
    pred_range = pred_range,
    xlim = xlim,
    mar = mar
  )
  # Add abaxial side to the same plot
  plot_gam_ci(
    data = df_abaxial,
    x = x,
    y = y,
    side = "abaxial",
    pred_range = pred_range,
    xlim = xlim,
    mar = mar,
    add = TRUE, # add to existing plot
    pch = 22 # Use different point shape for abaxial
  )
  dev.off()
})

# (G) - (L) RWC vs E
lapply(seq_along(spp), function(i) {
