# Load data
df_list <- list()
df_list[["tensile"]] <- read.csv("outputs/tables/tensile.csv")
categories <- c("bending", "tensile")
df <- add_tensile_stiffness(df_list[[categories[2]]])
df <- df[df$water_content >= 60, ] # filter data for water content >= 60%

df$I <- df$I * 1000 # convert unit of I
df$LMA <- df$dry_weight / df$area * 10^3 # calculate leaf mass per area (LMA) (g m-2)
df$density <- df$dry_weight / df$thickness # calculate leaf density (kg m-3)
df$E_per_density <- df$E.x / df$density # calculate E per density (MPa kg m-3)
df$E_per_density <- df$E_per_density * 10^3 # convert unit of E per density (MPa g m-3)

spp <- c(
  "ubayuri",
  "youshuyamagobou",
  "yurinoki",
  "niseakashia",
  "aoki",
  "nezumimochi"
) # species names
names(spp) <- c(
  "Cardiocrinum_cordatum",
  "Phytolacca_americana",
  "Liriodendron_tulipifera",
  "Robinia_pseudoacacia",
  "Aucuba_japonica",
  "Ligustrum lucidum"
)
pred_range <- c(60, 100)
xlim <- c(100, 60)
mar <- c(5, 5, 0.5, 0.5) # Margins

# Plot GAM with confidence intervals ----
output_dir <- "outputs/figures/figure_02"
if (!dir.exists(output_dir)) {
  dir.create(output_dir, recursive = TRUE)
}
side <- "tensile"
# (A) - (F) RWC vs EA
lapply(seq_along(spp), function(i) {
  sp <- spp[i]
  df_sp <- df[df$spp == sp, ]
  output_path <- file.path(
    output_dir,
    paste0(LETTERS[i], "_RWC_vs_EA_", names(spp)[i], ".svg")
  )
  svg(output_path, bg = "transparent")
  # Plot adaxial side
  plot_gam_ci(
    data = df_sp,
    x = "water_content",
    y = "EA",
    side = side,
    pred_range = pred_range,
    xlim = xlim,
    mar = mar
  )
  dev.off()
})

# (G) - (L) RWC vs E
lapply(seq_along(spp), function(i) {
  sp <- spp[i]
  df_sp <- df[df$spp == sp, ]
  output_path <- file.path(
    output_dir,
    paste0(LETTERS[6 + i], "_RWC_vs_E_", names(spp)[i], ".svg")
  )
  svg(output_path, bg = "transparent")
  # Plot adaxial side
  plot_gam_ci(
    data = df_sp,
    x = "water_content",
    y = "E.x",
    side = side,
    pred_range = pred_range,
    xlim = xlim,
    mar = mar
  )
  dev.off()
})

# (M) - (R) RWC vs Leaf tissue density

## Tensile stiffness ($E_tA$, N)

Cross-sectional area is `thick * width.x` (mm²), using the instrument dimensions
associated with `E.x` (MPa = N/mm²). The legacy tensile `EI` column is not used.

::::

:::: {.content-visible when-profile="ja"}

## 引張剛性（$E_tA$、N）

断面積は `thick * width.x`（mm²）で、`E.x`（MPa = N/mm²）に対応する試験機の寸法値を用います。従来の引張データの `EI` 列は用いません。
