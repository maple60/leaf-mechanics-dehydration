list_drop_adaxial <- lapply(seq_along(spp), function(i) {
  sp <- names(spp)[i]
  df_sp <- df_merged[df_merged$species == sp & df_merged$side == "adaxial", ]
  ei_100 <- df_sp$y[which.min(abs(df_sp$x - 100))]
  ei_80 <- df_sp$y[which.min(abs(df_sp$x - 80))]
  drop <- (ei_100 - ei_80) / ei_100 * 100
  df_drop <- data.frame(
    species = sp,
    side = "adaxial",
    ei_100 = ei_100,
    ei_80 = ei_80,
    drop = drop
  )
})

cex_pt <- 3 # Point size
pt_fill_alpha <- 0.5 # Transparency of points

# Include negative decreases (predicted stiffness increases) in the plot.

  "drop_tensile", "drop_adaxial", "drop_abaxial")])), na.rm = TRUE)
plot(
  df_wide$drop_tensile,
  df_wide$drop_adaxial,
  xlim = plot_limits,
  ylim = plot_limits,
  xlab = "",
  ylab = "",
  asp = 1,
  type = "n",
  xaxt = "n",
  yaxt = "n"
)

# Adaxial

  df_wide$drop_tensile,
  df_wide$drop_adaxial,
  pch = 21,
  cex = cex_pt,
  bg = adjustcolor(7, alpha.f = pt_fill_alpha),
  col = adjustcolor(7, alpha.f = pt_border_alpha)
)
# Abaxial
points(
  df_wide$drop_tensile,
  df_wide$drop_abaxial,
  pch = 22,

  col = adjustcolor(6, alpha.f = pt_border_alpha)

  drop_adaxial ~ drop_tensile,

regression_summary <- do.call(rbind, lapply(c("Adaxial", "Abaxial"), function(side) {
  model <- if (side == "Adaxial") model_adaxial else model_abaxial
  model_summary <- summary(model)
  data.frame(side = side, species = nobs(model), intercept = unname(coef(model)[1]),
    slope = unname(coef(model)[2]), R2 = model_summary$r.squared,
    adjusted_R2 = model_summary$adj.r.squared, p = coef(model_summary)[2, 4])
}))
